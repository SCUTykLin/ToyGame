cmake_minimum_required(VERSION 3.25)
project(upfps)
add_library(upfps STATIC)
file(GLOB_RECURSE SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
target_sources(upfps PRIVATE ${SOURCE})
target_include_directories(upfps PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(upfps PRIVATE olcEngin)
target_compile_definitions(upfps PRIVATE _UNICODE UNICODE)
