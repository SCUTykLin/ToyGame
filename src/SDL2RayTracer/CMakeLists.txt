cmake_minimum_required(VERSION 3.25)
project(sdl2raytracer)
add_library(sdl2raytracer STATIC)
file(GLOB_RECURSE SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
target_sources(sdl2raytracer PRIVATE ${SOURCE})
target_include_directories(sdl2raytracer PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(sdl2raytracer PRIVATE SDL2::SDL2main SDL2::SDL2-static SDL2_image SDL2_ttf glm::glm-header-only)
