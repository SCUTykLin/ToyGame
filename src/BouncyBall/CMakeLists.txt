cmake_minimum_required(VERSION 3.25)
project(bouncyball)
add_library(bouncyball)
file(GLOB_RECURSE SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
target_sources(bouncyball PRIVATE ${SOURCE})
target_include_directories(bouncyball PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/)
target_link_libraries(bouncyball PRIVATE SDL2::SDL2main SDL2::SDL2-static SDL2_image)
